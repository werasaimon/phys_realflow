# Optional engine references, isolated from the SDK dependency graph.
cmake_minimum_required(VERSION 3.21)
project(RigidEngineReferences LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(RF_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(RF_BUILD "${RF_SOURCE}/build-core" CACHE PATH "Prebuilt PhysRealFlow libraries")
find_package(Threads REQUIRED)
find_package(Jolt CONFIG QUIET)
find_package(Bullet QUIET)
find_package(box3d CONFIG QUIET)

function(reference_warnings name)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()
function(reference_target name source)
    add_executable(${name} ../${source})
    target_include_directories(${name} PRIVATE "${RF_SOURCE}" "${RF_SOURCE}/src")
    target_link_libraries(${name} PRIVATE
        "${RF_BUILD}/${CMAKE_STATIC_LIBRARY_PREFIX}rfsamples${CMAKE_STATIC_LIBRARY_SUFFIX}"
        "${RF_BUILD}/${CMAKE_STATIC_LIBRARY_PREFIX}rfcore${CMAKE_STATIC_LIBRARY_SUFFIX}" Threads::Threads)
    reference_warnings(${name})
endfunction()

if(Jolt_FOUND)
    reference_target(rf_jolt_reference JoltTorusReference.cpp)
    target_link_libraries(rf_jolt_reference PRIVATE Jolt::Jolt)
endif()
if(BULLET_FOUND)
    reference_target(rf_bullet_reference BulletTorusReference.cpp)
    target_include_directories(rf_bullet_reference SYSTEM PRIVATE ${BULLET_INCLUDE_DIRS})
    target_link_libraries(rf_bullet_reference PRIVATE ${BULLET_LIBRARIES})
    reference_target(rf_simbenchmark_bouncing SimBenchmarkBouncing.cpp)
    target_include_directories(rf_simbenchmark_bouncing SYSTEM PRIVATE ${BULLET_INCLUDE_DIRS})
    target_link_libraries(rf_simbenchmark_bouncing PRIVATE ${BULLET_LIBRARIES})
    if(Jolt_FOUND)
        target_compile_definitions(rf_simbenchmark_bouncing PRIVATE RF_REFERENCE_JOLT)
        target_link_libraries(rf_simbenchmark_bouncing PRIVATE Jolt::Jolt)
    endif()
    if(box3d_FOUND)
        target_compile_definitions(rf_simbenchmark_bouncing PRIVATE RF_REFERENCE_BOX3D)
        target_link_libraries(rf_simbenchmark_bouncing PRIVATE box3d::box3d)
    endif()
endif()
if(box3d_FOUND)
    add_library(rf_box3d_backend STATIC Box3dRigidBackend.cpp)
    target_include_directories(rf_box3d_backend PUBLIC "${RF_SOURCE}/src" "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(rf_box3d_backend PRIVATE box3d::box3d
        "${RF_BUILD}/${CMAKE_STATIC_LIBRARY_PREFIX}rfcore${CMAKE_STATIC_LIBRARY_SUFFIX}" Threads::Threads)
    reference_warnings(rf_box3d_backend)
    reference_target(rf_box3d_reference Box3dTorusReference.cpp)
    target_link_libraries(rf_box3d_reference PRIVATE rf_box3d_backend)
    reference_target(rf_box3d_backend_checks Box3dBackendChecks.cpp)
    target_sources(rf_box3d_backend_checks PRIVATE SatCcd.cpp)
    target_link_libraries(rf_box3d_backend_checks PRIVATE rf_box3d_backend)
    enable_testing()
    add_test(NAME box3d_backend_si COMMAND rf_box3d_backend_checks)
endif()
if(NOT Jolt_FOUND AND NOT BULLET_FOUND AND NOT box3d_FOUND)
    message(FATAL_ERROR "Install at least one reference engine or set CMAKE_PREFIX_PATH")
endif()
